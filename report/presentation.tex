\section{Contexte et problématique}

\begin{frame}{D'où viennent les bases orientées colonnes ?}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{itemize}
        \item Le relationnel (Codd, 1970) repose sur des jointures et des transactions, sur un serveur unique.
        \item Un serveur ne grandit pas indéfiniment : avec des millions d'écritures continues, on répartit les données sur de nombreuses machines ordinaires.
        \item Il faut alors continuer à fonctionner quand l'une d'elles tombe en panne.
      \end{itemize}
    \end{column}
    \begin{column}{0.44\textwidth}
      \centering
      \begin{tikzpicture}[font=\footnotesize,
          ev/.style={circle,fill=accent,inner sep=2pt},
          lb/.style={anchor=west,align=left,text width=4.2cm}]
        \draw[gris,thick] (0,0) -- (0,-3.6);
        \node[ev] at (0,0) {};    \node[lb] at (0.25,0)    {\textbf{2006} Google : Bigtable\\{\color{gris}lignes triées, familles de colonnes}};
        \node[ev] at (0,-1.2) {}; \node[lb] at (0.25,-1.2) {\textbf{2007} Amazon : Dynamo\\{\color{gris}clé-valeur sans maître, disponible}};
        \node[ev] at (0,-2.4) {}; \node[lb] at (0.25,-2.4) {\textbf{2008} Facebook : Cassandra\\{\color{gris}modèle Bigtable + architecture Dynamo}};
        \node[ev] at (0,-3.6) {}; \node[lb] at (0.25,-3.6) {\textbf{2010} Projet Apache\\{\color{gris}aujourd'hui Cassandra 5}};
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Problématique}
  \vfill
  \begin{block}{}
    \centering\large
    Comment les bases orientées colonnes gèrent-elles efficacement\\des volumes massifs de données distribuées,\\et quels compromis impliquent-elles ?
  \end{block}
  \vspace{0.6cm}
  \begin{itemize}
    \item Exemple suivi tout au long de l'exposé : un réseau de capteurs de température et d'humidité.
    \item Avec ces bases, on part des requêtes que l'application devra exécuter ; en relationnel, on part des données.
  \end{itemize}
  \vfill
\end{frame}

\section{Le modèle de données}

\begin{frame}{Deux sens du mot \og colonne \fg{}}
  \centering
  \resizebox{0.92\textwidth}{!}{%
  \begin{tikzpicture}[font=\footnotesize,

    \node[b,fill=accent!8,minimum width=3.0cm,right=0cm of w1,font=\scriptsize] (w2) {ts1 : temp=27.4};
    \node[b,fill=accent!8,minimum width=3.0cm,right=0cm of w2,font=\scriptsize] (w3) {ts1 : hum=61};
    \node[b,fill=accent!8,minimum width=3.0cm,right=0cm of w3,font=\scriptsize] (w4) {ts2 : temp=27.6};
    \node[b,fill=accent!8,minimum width=3.0cm,right=0cm of w4,font=\scriptsize] {ts2 : hum=60};
  \end{tikzpicture}}
  \vspace{0.3cm}
  \begin{itemize}
    \item Le stockage colonnaire (ClickHouse, Redshift, Parquet) sert l'analyse : peu de colonnes lues sur beaucoup de lignes.
    \item Le modèle Wide-Column (Bigtable, Cassandra, HBase, ScyllaDB) sert la distribution et l'écriture massive. C'est notre sujet.
  \end{itemize}
\end{frame}

\begin{frame}{Le modèle de Bigtable}
  \vspace{0.2cm}
  \begin{center}
    \large
    $(\textit{clé de ligne},\ \textit{famille:colonne},\ \textit{horodatage}) \longmapsto \textit{valeur}$
  \end{center}
  \vspace{0.3cm}
  \begin{itemize}
    \item Les lignes sont triées par clé : les lectures par plage sont efficaces.
    \item Les familles de colonnes sont déclarées à l'avance ; les colonnes s'ajoutent librement à l'intérieur.
    \item Une cellule peut garder plusieurs versions, distinguées par leur horodatage.
    \item La table est éparse : une colonne sans valeur ne coûte rien, deux lignes peuvent avoir des colonnes différentes.
  \end{itemize}
\end{frame}

\begin{frame}{Le vocabulaire de Cassandra}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \small
      \begin{tabularx}{\linewidth}{@{}l>{\raggedright\arraybackslash}X@{}}
        \toprule
        Élément & Rôle \\
        \midrule
        Cluster & Ensemble des nœuds \\
        Keyspace & Regroupe les tables, fixe la réplication \\
        Table & Partitions décrites par un schéma \\
        Partition & Lignes de même clé de partition ; unité de répartition \\
        Ligne & Repérée par sa clé de clustering \\
        Cellule & Valeur + horodatage (+ TTL) \\
        \bottomrule
      \end{tabularx}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Clé primaire en deux parties}
        \small
        \cq{PRIMARY KEY ((sensor_id, day), ts)}\par\medskip
        La clé de partition \cq{(sensor_id, day)} choisit les nœuds qui stockent la donnée.\par\medskip
        La clé de clustering \cq{ts} trie les lignes dans la partition.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

  Une partition peut contenir un très grand nombre de lignes, triées par la clé de clustering.
  \vspace{0.5cm}

  \end{itemize}
\end{frame}

  \small
  \begin{tabularx}{\textwidth}{@{}>{\raggedright\arraybackslash}p{2.8cm}>{\raggedright\arraybackslash}X>{\raggedright\arraybackslash}X>{\raggedright\arraybackslash}X@{}}

\section{Architecture distribuée}

\begin{frame}{Partitionnement et réplication}
  \begin{columns}[c]
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item Pas de serveur maître : tous les nœuds ont le même rôle.
        \item Murmur3 transforme la clé de partition en un \emph{token}, entre $-2^{63}$ et $2^{63}-1$.
        \item Chaque nœud possède des plages de l'anneau (\emph{vnodes}) ; ajouter un nœud ne déplace qu'une fraction des données.
        \item Le facteur de réplication (RF) fixe le nombre de copies. Aucune copie n'est principale.
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \centering
      \begin{tikzpicture}[font=\small,
        nd/.style={circle,draw=gris,minimum size=0.85cm,fill=white,thick},
        rep/.style={circle,draw=accent,minimum size=0.85cm,fill=accent!15,thick}]
        \draw[gris!50,thick] (0,0) circle (1.9);
        \node[nd] (A) at (90:1.9) {A};
        \node[rep] (B) at (30:1.9) {B};
        \node[rep] (C) at (-30:1.9) {C};
        \node[rep] (D) at (-90:1.9) {D};
        \node[nd] (E) at (-150:1.9) {E};
        \node[nd] (F) at (150:1.9) {F};
        \node[draw,fill=accent,text=white,diamond,inner sep=1pt] (K) at (58:1.9) {\scriptsize K};
        \draw[->,thick,accent] (K) -- (B);
        \node[font=\footnotesize,align=center,text=gris] at (0,-2.75) {Clé K dans la plage de B ; avec RF = 3,\\copies sur B, C et D};
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Cohérence ajustable}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \small
      \begin{tabularx}{\linewidth}{@{}l>{\raggedright\arraybackslash}X@{}}
        \toprule
        Niveau & Copies qui doivent répondre \\
        \midrule
        \cq{ONE} & Une seule : le plus rapide \\
        \cq{QUORUM} & La majorité : 2 sur 3 si RF = 3 \\
        \cq{LOCAL_QUORUM} & La majorité dans le datacenter local \\
        \cq{ALL} & Toutes : une panne bloque la requête \\
        \bottomrule
      \end{tabularx}
    \end{column}
    \begin{column}{0.41\textwidth}
      \begin{block}{Lire la dernière écriture}
        \centering
        $R + W > RF$\par\medskip
        \small Avec RF = 3, écrire et lire en \cq{QUORUM} : $2 + 2 = 4 > 3$.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.5cm}
  \begin{itemize}
    \item Théorème CAP : lors d'une coupure réseau, on renonce à la cohérence ou à la disponibilité. Cassandra privilégie la disponibilité par défaut.
    \item PACELC : même sans panne, il reste un compromis entre latence et cohérence.
  \end{itemize}
\end{frame}

\begin{frame}{Tolérance aux pannes}
  Cassandra n'a pas de point de défaillance unique. Quatre mécanismes se relaient :
  \vspace{0.3cm}
  \begin{itemize}
    \item le \emph{gossip} : chaque nœud échange avec quelques voisins l'état de tous les autres et détecte les pannes ;
    \item le \emph{hinted handoff} : un nœud garde les écritures destinées à un nœud absent et les lui transmet à son retour (3 heures par défaut) ;
    \item le \emph{read repair} : si deux copies diffèrent à la lecture, la plus récente est propagée ;
    \item l'\emph{anti-entropy repair} : \cq{nodetool repair} compare les copies par arbres de Merkle.
  \end{itemize}
\end{frame}

\begin{frame}{Chemin d'écriture et de lecture}
  \centering
  \begin{tikzpicture}[font=\footnotesize,
    box/.style={draw=gris,rounded corners,minimum height=0.8cm,minimum width=2.0cm,align=center,fill=white},
    arr/.style={-{Stealth},thick,accent}]
    \node[box,fill=fond] (cl) {Client};
    \node[box,right=0.6cm of cl,fill=fond] (co) {Nœud\\coordinateur};
    \node[box,above right=0.1cm and 0.7cm of co,fill=accent!12] (log) {Commit log\\(disque, ajout seul)};
    \node[box,below right=0.1cm and 0.7cm of co,fill=accent!12] (mem) {Memtable\\(mémoire)};
    \node[box,right=0.9cm of mem,fill=accent!25] (ss) {SSTable\\(fichier immuable)};
    \node[box,right=0.6cm of ss,fill=fond] (cp) {Compaction};
    \draw[arr] (cl) -- (co);
    \draw[arr] (co) -- (log);
    \draw[arr] (co) -- (mem);
    \draw[arr] (mem) -- node[above]{flush} (ss);
    \draw[arr] (ss) -- (cp);
  \end{tikzpicture}
  \vspace{0.4cm}
  \begin{itemize}
    \item Le client est confirmé dès que le commit log et la memtable sont à jour : aucune lecture préalable, aucune modification sur place (LSM-tree).
    \item À la lecture, la memtable et les SSTables sont fusionnées ; la cellule la plus récente l'emporte.
    \item Un filtre de Bloom signale qu'une SSTable ne contient pas la partition et évite de la lire.
    \item Une suppression écrit une \emph{tombstone}, effacée à la compaction après \cq{gc_grace_seconds} (10 jours).
  \end{itemize}
\end{frame}

\section{Performances, systèmes et usages}

\begin{frame}{Ce que l'on gagne, ce que l'on paie}
  \small
  \begin{tabularx}{\textwidth}{@{}>{\raggedright\arraybackslash}X>{\raggedright\arraybackslash}X@{}}
    \toprule
    Avantage & Contrepartie \\
    \midrule
    Montée en charge par ajout de machines, haute disponibilité & Cohérence forte non automatique : il faut choisir les niveaux \\
    Écritures très rapides (LSM-tree) & Lectures plus coûteuses, compaction en arrière-plan \\
    Aucune jointure à exécuter à la lecture & Duplication : une table par type de requête \\
    Schéma souple & Une requête non prévue est coûteuse ou refusée \\
    \bottomrule
  \end{tabularx}
  \vspace{0.5cm}
  \normalsize
  \begin{itemize}
    \item La complexité passe de la lecture (jointures, agrégations) à la conception des tables et à l'écriture.
    \item Une mauvaise conception se corrige rarement par une requête plus astucieuse : il faut souvent recréer la table.
  \end{itemize}
\end{frame}

\begin{frame}{Principaux systèmes}
  \small
  \begin{tabularx}{\textwidth}{@{}>{\raggedright\arraybackslash}p{2.3cm}>{\raggedright\arraybackslash}X>{\raggedright\arraybackslash}X>{\raggedright\arraybackslash}X@{}}
    \toprule
     & Cassandra & HBase & ScyllaDB \\
    \midrule
    Langage & Java & Java & C++ \\
    Architecture & Pair-à-pair & Maître + régions, HDFS & Pair-à-pair, un fragment par cœur \\
    Interface & CQL & API HBase, shell & CQL (compatible Cassandra) \\
    Cohérence & Ajustable & Forte par ligne & Ajustable \\
    Point fort & Disponibilité, multi-datacenter & Intégration Hadoop et Spark & Performance par machine \\
    Limite & Réglage de la JVM, compaction & Déploiement complexe & Écosystème plus restreint \\
    \bottomrule
  \end{tabularx}
  \vspace{0.4cm}
  \normalsize
  Google Cloud Bigtable, service géré issu du système d'origine, s'utilise avec l'API de HBase.
\end{frame}

\begin{frame}{Cas d'utilisation}

\section{Étude de cas : capteurs IoT avec Cassandra}

\begin{frame}{Partir des requêtes, pas des données}
  Des capteurs envoient une mesure (température, humidité, batterie) toutes les 10 secondes.
  \vspace{0.3cm}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \small
      \begin{tabularx}{\linewidth}{@{}l>{\raggedright\arraybackslash}X@{}}
        \toprule
        N° & Requête métier \\
        \midrule
        Q1 & Dernières mesures d'un capteur pour un jour \\
        Q2 & Mesures entre deux heures \\
        Q3 & Dernière mesure de chaque capteur \\
        Q4 & Alertes ($>$ 35\,\textdegree C) d'une zone pour un jour \\
        Q5 & Indicateurs d'une journée \\
        Q6 & Évolution des indicateurs sur une période \\
        \bottomrule
      \end{tabularx}
    \end{column}
    \begin{column}{0.46\textwidth}
      \small
      \begin{tabularx}{\linewidth}{@{}>{\raggedright\arraybackslash}X l@{}}
        \toprule
        Table & Sert \\
        \midrule
        \cq{readings_by_sensor_day} & Q1, Q2, Q5 \\
        \cq{last_reading_by_sensor} & Q3 \\
        \cq{alerts_by_zone_day} & Q4 \\
        \cq{daily_stats_by_sensor} & Q6 \\
        \bottomrule
      \end{tabularx}
      \vspace{0.3cm}
      Six requêtes, quatre tables : une même mesure est écrite plusieurs fois.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{La table principale}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
\begin{lstlisting}[language=CQL]
CREATE TABLE readings_by_sensor_day (
  sensor_id text,
  day date,
  ts timestamp,
  temperature double,
  humidity double,
  battery_pct int,
  PRIMARY KEY ((sensor_id, day), ts)
) WITH CLUSTERING ORDER BY (ts DESC)
  AND default_time_to_live = 7776000
  AND compaction = {
    'class': 'TimeWindowCompactionStrategy',
    'compaction_window_unit': 'DAYS',
    'compaction_window_size': 1 };
\end{lstlisting}
    \end{column}
    \begin{column}{0.44\textwidth}
      \small
      \begin{itemize}
        \item Le jour dans la clé de partition borne sa taille : $6 \times 60 \times 24 = 8\,640$ lignes par jour au lieu de 3,15 millions par an (\emph{bucketing}).
        \item \cq{ts DESC} : Q1 lit les premières lignes, sans tri.
        \item Expiration après 90 jours et compaction par fenêtres de temps.
        \item Keyspace répliqué sur 3 nœuds (RF = 3).
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Requêtes avec la clé de partition}
  \begin{columns}[c]
    \begin{column}{0.58\textwidth}
      \centering
      \includegraphics[width=\linewidth]{fig6a_q1_q2.png}
    \end{column}
    \begin{column}{0.38\textwidth}
      \small
      \begin{itemize}
        \item Q1 renvoie les mesures dans l'ordre décroissant, sans tri.
        \item Q2 lit une plage contiguë de la partition.
        \item Cassandra sait quels nœuds interroger, car la requête donne la clé de partition complète.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Une requête non prévue est refusée}
  \centering
  \includegraphics[width=0.9\textwidth]{fig7_allow_filtering.png}
  \vspace{0.4cm}
  \begin{itemize}
    \item Sans clé de partition, il faudrait parcourir toutes les partitions de tous les nœuds.
    \item \cq{ALLOW FILTERING} force l'exécution : négligeable sur quelques lignes, très lourd sur des millions.
    \item La bonne réponse est une table conçue pour la question, ici \cq{alerts_by_zone_day}.
  \end{itemize}
\end{frame}

\begin{frame}{Réplication, partitionnement et panne}
  \begin{columns}[c]
    \begin{column}{0.55\textwidth}
      \centering
      \includegraphics[width=\linewidth]{fig10_replication_partitionnement.png}
    \end{column}
    \begin{column}{0.41\textwidth}
      \small
      \begin{itemize}
        \item RF = 3 sur 3 nœuds : chaque nœud possède 100\,\% des données de \cq{iot}.
        \item RF = 1 : chaque capteur est confié au nœud que désigne le hachage de son identifiant.
        \item Après \cq{docker stop cass3}, Q1 réussit encore en \cq{QUORUM} : deux copies sur trois suffisent.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\section{Mesures sur dix ans : Cassandra face à PostgreSQL}

\begin{frame}{Protocole}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item 50 capteurs, 5 zones, une mesure toutes les 15 minutes.
        \item 1\,752\,000 mesures par an ; \textbf{17\,529\,600} sur dix ans (2016 à 2026).
        \item Mêmes données dans les deux bases, générées avec une graine fixe.
        \item Mesures après 1, 2, 3, 5 et 10 ans d'historique.
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{itemize}
        \item Cassandra 5.0.9, un nœud, même modèle que l'étude de cas.
        \item PostgreSQL 15.19, schéma normalisé, clé primaire \cq{(sensor_id, ts)}.
        \item Médiane de 50 exécutions ; lectures à froid après redémarrage et vidage des caches.
        \item Scripts et résultats bruts fournis avec le rapport.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Volume stocké}
  \centering
  \includegraphics[width=0.82\textwidth]{fig13_volumes.png}
  \vspace{0.3cm}
  \small
  \begin{tabular}{@{}lrrrrr@{}}
    \toprule
    Historique & 1 an & 2 ans & 3 ans & 5 ans & 10 ans \\
    \midrule
    Cassandra (Mo) & 35 & 69 & 104 & 174 & 349 \\
    PostgreSQL, table + index (Mo) & 175 & 350 & 526 & 876 & 1\,755 \\
    \bottomrule
  \end{tabular}
  \vspace{0.2cm}
  \normalsize
  \par Cinq fois moins de place pour Cassandra : compression LZ4, clé de partition écrite une seule fois.
\end{frame}

\begin{frame}{Temps de lecture selon le volume}
  \centering
  \includegraphics[width=\textwidth]{fig11_latence_volume.png}
  \vspace{0.2cm}
  \begin{itemize}
    \item Cassandra reste entre 3 et 5\,ms de 1 à 10 ans : la partition contient toujours 96 lignes.
    \item PostgreSQL lit plus vite sur une seule machine, mais Q1 ancien passe de 0,37 à 1,36\,ms.
    \item Q6, prévue d'avance : 7\,ms (table précalculée) contre 22\,ms (agrégation à chaque appel).
  \end{itemize}
\end{frame}

\begin{frame}{Retrouver une journée de 2016}
  \begin{columns}[c]

\section{Étude comparative}

\begin{frame}{Relationnel et Wide-Column, critère par critère}
  \footnotesize
  \begin{tabularx}{\textwidth}{@{}>{\raggedright\arraybackslash}p{3.6cm}>{\raggedright\arraybackslash}X>{\raggedright\arraybackslash}X@{}}
    \toprule
    Critère & Relationnel (PostgreSQL) & Wide-Column (Cassandra) \\
    \midrule
    Conception & À partir des données & À partir des requêtes \\
    Schéma de l'étude de cas & 2 tables, sans duplication & 4 tables, mesures dupliquées \\
    Jointures & À la lecture & Remplacées par la duplication \\
    Transactions & ACID sur plusieurs tables & Lot atomique, transactions légères \\
    Montée en charge & Serveur plus puissant & Ajout de nœuds \\
    Lecture par clé (mesurée) & 0,3 à 1,6\,ms, en hausse & 3 à 5\,ms, constante sur 10 ans \\
    Écritures unitaires (mesurées) & 305 /s & 2\,504 /s \\
    Espace disque, 10 ans (mesuré) & 1\,755\,Mo & 349\,Mo \\
    Question imprévue (mesurée) & Exécutée, 16 à 24\,s & Échec à 12\,s \\
    Nouveau besoin & Nouvelle requête & Nouvelle table + migration \\
    \bottomrule
  \end{tabularx}
\end{frame}

\begin{frame}{Études de cas comparatives}
  \small
  \begin{tabularx}{\textwidth}{@{}>{\raggedright\arraybackslash}p{2.6cm}>{\raggedright\arraybackslash}X>{\raggedright\arraybackslash}X>{\raggedright\arraybackslash}p{2.2cm}@{}}
    \toprule
    Cas & Conception relationnelle & Conception Wide-Column & Retenu \\
    \midrule
    Capteurs IoT & \cq{readings(sensor_id, ts)} indexée & \cq{((sensor_id, day), ts)} & Wide-Column \\
    Messagerie & \cq{messages} indexée par conversation & \cq{((conversation_id, mois), date)} & Wide-Column \\
    Commandes en ligne & Stock, paiement, lignes dans une transaction ACID & Plusieurs tables sans transaction commune & Relationnel \\
    Journaux applicatifs & Table qui grossit, purge par \cq{DELETE} & \cq{((service, heure), date)} + TTL & Wide-Column \\
    \bottomrule
  \end{tabularx}
  \vspace{0.4cm}
  \normalsize
  \begin{itemize}
    \item Le choix dépend de la nature des accès plus que de la taille.
    \item Requêtes connues, écritures nombreuses, distribution : Wide-Column.
    \item Transactions entre entités, questions changeantes : relationnel. Beaucoup de systèmes combinent les deux.
  \end{itemize}
\end{frame}

\section{Du relationnel au Wide-Column}

\begin{frame}{Peut-on passer du relationnel au Wide-Column ?}
  Oui. Il n'y a pas de conversion automatique : on refait le schéma en partant des requêtes, puis on recopie les données.
  \vspace{0.3cm}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{enumerate}
        \item Inventorier les requêtes de l'application (code, \cq{pg_stat_statements}).
        \item Garder le modèle conceptuel (entités, associations).
        \item Concevoir une table par requête fréquente.
        \item Choisir les clés : égalités en partition, plages et tri en clustering, période pour borner.
      \end{enumerate}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{enumerate}\setcounter{enumi}{4}
        \item Fixer TTL, compaction, réplication.
        \item Migrer l'historique : jointures et agrégations faites à l'export.
        \item Vérifier : comptages et mêmes réponses dans les deux bases.
        \item Basculer : double écriture ou capture des changements (Debezium, Kafka), puis lectures sur Cassandra.
      \end{enumerate}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Règles de correspondance}
  \small
  \begin{tabularx}{\textwidth}{@{}>{\raggedright\arraybackslash}p{4.6cm}>{\raggedright\arraybackslash}X@{}}
    \toprule
    Relationnel & Wide-Column (Cassandra) \\
    \midrule
    Table normalisée & Plusieurs tables dénormalisées, une par requête \\
    Jointure, clé étrangère & Colonnes recopiées dans la table qui en a besoin \\
    Association 1-N & Lignes filles dans la partition du parent, triées par clustering \\
    Association N-N & Deux tables, une par sens de lecture \\
    \cq{GROUP BY} fréquent & Table précalculée \\
    Index secondaire & Table dédiée à la requête \\
    Contrainte d'intégrité & Contrôle par l'application \\
    Transaction sur plusieurs tables & \cq{BATCH} logué ou transaction légère (\cq{IF NOT EXISTS}) \\
    Identifiant auto-incrémenté & \cq{uuid} ou \cq{timeuuid} \\
    \bottomrule
  \end{tabularx}
\end{frame}

\begin{frame}[fragile]{Application à notre schéma}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \small
      \begin{tabularx}{\linewidth}{@{}>{\raggedright\arraybackslash}X>{\raggedright\arraybackslash}p{2.3cm}@{}}
        \toprule
        Table Cassandra & Export SQL \\
        \midrule
        \cq{readings_by_sensor_day} & \cq{ts::date} \\
        \cq{alerts_by_zone_day} & jointure \cq{sensors} \\
        \cq{daily_stats_by_sensor} & \cq{GROUP BY} \\
        \cq{last_reading_by_sensor} & \cq{LATERAL} \\
        \bottomrule

      \vspace{0.3cm}
      Le travail que PostgreSQL faisait à chaque lecture est fait une seule fois, à l'export.
    \end{column}
    \begin{column}{0.5\textwidth}
\begin{lstlisting}[language=SQL]
-- psql : export de la table des alertes
\copy (SELECT s.zone, r.ts::date, ...,
         r.sensor_id, r.temperature
       FROM readings r JOIN sensors s
         ON s.sensor_id = r.sensor_id
       WHERE r.temperature > 35)
  TO '/tmp/alerts.csv' WITH (FORMAT csv)
\end{lstlisting}
\begin{lstlisting}[language=CQL]
// cqlsh : import
COPY iot_migre.alerts_by_zone_day
  (zone, day, ts, sensor_id, temperature)
  FROM '/tmp/alerts.csv';
\end{lstlisting}
      \scriptsize\color{gris} Procédure complète : dossier \texttt{migration/} (\texttt{migrer.sh}).
    \end{column}
  \end{columns}
\end{frame}

\section{Conclusion}

\begin{frame}{Conclusion}
  \begin{itemize}
    \item Les bases Wide-Column stockent de très grands volumes arrivant en continu, sur des machines ordinaires, sans point de défaillance unique.
    \item Elles reposent sur le partitionnement par hachage, la réplication sans maître, la cohérence réglable et l'écriture en ajout seul.
    \item En retour, il faut connaître les requêtes avant de créer les tables, accepter la duplication et renoncer aux jointures.
    \item Sur dix ans de mesures, le temps de lecture de Cassandra n'a pas bougé avec le volume ; la requête imprévue a fini par échouer.
    \item En pratique, on associe souvent une base Wide-Column pour l'ingestion et un entrepôt colonnaire pour l'analyse.
  \end{itemize}
\end{frame}
